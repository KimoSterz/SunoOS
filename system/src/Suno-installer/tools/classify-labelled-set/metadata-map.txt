# The distributions' own metadata, as a second opinion on the classifier (curate.py).
# Each line: a pattern, the class it means, and why. A package whose sources give
# different classes has no opinion ("conflict"); one with no matching line has none.
# The comparison is with the classifier's default (DEFAULT-RULE), by kind.
#
# [appstream]  AppStream component types (Ubuntu DEP-11, Fedora appstream-data XML)
# [debtags]    Debian debtags (from Debian trixie's Packages Tag fields, by package name;
#              Ubuntu has no tags of its own). Rules are tried in order on the tag set;
#              "a & b" needs both, "a | b" either; fnmatch patterns.
# [comps]      Fedora comps group ids (fnmatch), for the packages a group lists

[appstream]
desktop-application  application     # a launcher a person opens: rule 6
desktop              application     # the older name of desktop-application
console-application  workspace       # a terminal program, typed: rule 7
addon                application     # extends one application: added to its tree (add_on_of), ADR Applications 8
font                 device-add-on   # fonts for the person: rule 5

inputmethod          device-add-on   # an input-method engine: rule 4
